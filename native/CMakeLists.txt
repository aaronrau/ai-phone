cmake_minimum_required(VERSION 3.22)
project(receptionist_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Inference is unusably slow without optimisation, so debug builds optimise too.
set(CMAKE_C_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
  message(FATAL_ERROR "Run tool/setup.py --native to provision the pinned llama.cpp source outside Git")
endif()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
foreach(INFERENCE_TARGET llama ggml ggml-base ggml-cpu)
  if(TARGET ${INFERENCE_TARGET})
    target_compile_options(${INFERENCE_TARGET} PRIVATE -O3)
  endif()
endforeach()

add_library(receptionist_llm SHARED llm_jni.cpp)
target_link_libraries(receptionist_llm PRIVATE llama android log)
target_compile_options(receptionist_llm PRIVATE -O3)

add_subdirectory(pocket)
